/** Fixed-point square root. sqrt(a) in Fx equals isqrt(a * FX_ONE). */
export function sqrt(a: Fx): Fx {
  if (a <= 0) return 0;
  return isqrt(a * FX_ONE) | 0;
}

/**
 * Length of a two-dimensional vector.
 *
 * The squares are summed as raw integer products before a single shift back to Q16.16, rather than being
 * computed with mul(). That is the entire reason this exists as a helper: mul() rounds each product
 * individually, so `sqrt(add(mul(x,x), mul(y,y)))` compounds two roundings before the square root sees them.
 * For near-equal components that can land on a different integer on a different platform, and a one-unit
 * difference in an angle feeds straight into a state hash and fails verification.
 *
 * Inputs are Q16.16, so each square is at most about 2^62 before the shift, inside the exact range of a
 * double for the magnitudes this simulation uses.
 */
export function length2(x: Fx, y: Fx): Fx {
  if (x === 0 && y === 0) return 0;
  const sum = x * x + y * y;
  // isqrt of the un-shifted sum gives the length already in Q16.16: sqrt(v * 2^32) === sqrt(v) * 2^16.
  return isqrt(sum) | 0;
}

/**
 * Length of a three-dimensional vector. Same reasoning as length2.
 */
export function length3(x: Fx, y: Fx, z: Fx): Fx {
  if (x === 0 && y === 0 && z === 0) return 0;
  return isqrt(x * x + y * y + z * z) | 0;
}
